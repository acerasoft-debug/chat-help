# Account, cart, session pages and generated downloads are NOT blocked here any more
# (Search Console, 12 Oct 2026: "Indexed, though blocked by robots.txt"). Hundreds of
# public pages link to /login?back=... and /cart, so Google indexed those URLs without
# being allowed to read them -- and a page it cannot read cannot tell it "noindex".
# Each of them now answers with `X-Robots-Tag: noindex` (and a matching meta tag), which
# drops them from the index for good. /admin stays noindex,nofollow the same way.

Disallow: /data/
Disallow: /vendor/
Disallow: /sheets/

# /uploads holds two very different things: product photography, and line sheets
# a seller uploaded (sheet_*.xlsx/xls/csv), which carry their own trade pricing.
# Blocking the whole folder kept the sheets private but also made every og:image
# and every Product-schema image uncrawlable, so no listing could earn an image
# in search. Only real image files are opened up; everything else stays closed.
Disallow: /uploads/
